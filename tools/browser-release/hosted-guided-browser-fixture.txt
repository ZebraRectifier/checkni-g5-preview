
test('exact spoken request keeps household and 2 kg quantity',async({page,context})=>{
 await context.route('**/*',route=>['127.0.0.1','localhost'].includes(new URL(route.request().url()).hostname)?route.continue():route.abort());
 await page.route('**/rest/v1/retail_catalog_categories**',route=>route.fulfill({json:[]}));
 await page.route('**/rest/v1/retail_catalog_items**',route=>route.fulfill({json:products.map(row=>({...row,observed_at:new Date().toISOString()}))}));
 await page.goto('/?comparison=observed&catalog=retail#shop');await selectStore(page);
 const status=page.locator('#basket-proposal-status');
 await submit(page,'Собери корзину для 3 человек на 14 дней');await expect(status).toContainText('общий бюджет');
 await submit(page,'не знаю');
 await submit(page,'Хочу 2 кг фарша, ограничений нет');
 await expect(status).toContainText('Фарш: 2 кг');
 await status.getByRole('button',{name:'Подобрать товары и цены',exact:true}).click();
 const row=page.locator('#hori-shopping-results').getByRole('table').getByRole('row').filter({hasText:'Фарш'}).first();
 await expect(row).toContainText('Запрос: 2 кг');
 await expect(row).toContainText('4 × 500 г');
 await status.getByRole('button',{name:/Добавить (найденное|показанное)/u}).click();
 const basket=await page.evaluate(()=>JSON.parse(localStorage.getItem('checkni.surface.basket.v1')??'[]'));
 // 3 people × 14 days × 200 ml => nine one-litre packages.
 expect(basket.find(item=>item.id.endsWith(':milk')).quantity).toBe(9);
});
